/** 策略列表 GET /api/v1/sqlaudit/audit_strategic/list_audit_strategic */
export async function listAuditStrategic(
  // 叠加生成的Param类型 (非body参数swagger默认没有生成对象)
  params: API.listAuditStrategicParams,
  options?: { [key: string]: any },
) {
  return request<API.ResultIPageAuditStrategicResponse>(
    '/api/v1/sqlaudit/audit_strategic/list_audit_strategic',
    {
      method: 'GET',
      params: {
        // pageNum has a default value: 1
        pageNum: '1',
        // pageSize has a default value: 10
        pageSize: '10',
        ...params,
      },
      ...(options || {}),
    },
  );
}

/**
 * @description 所有审核策略
 * @returns 审核策略列表
 * @returns 加载状态
 */
export function useAuditStrategicList() {
  const {
    data = {},
    loading,
    run: runAuditStrategic,
  } = useRequest(
    (realType = 1, auditSubType = 1) =>
      //@ts-ignore
      listAuditStrategic({ pageNum: 1, pageSize: 1000000, realType, auditSubType }),  //listAuditStrategic 是一个函数  
    {
      manual: true,
    },
  );
  return {
    runAuditStrategic,
    auditStrategicList: data?.records || [],
    auditStrategicLoading: loading,
  };
}
